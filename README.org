#+TITLE: FastMARC
#+OPTIONS: toc:nil num:nil

High-performance MARC record reader with dual indexing and field hooks.

Built with Cython and memory-mapped I/O. Fully compatible with ~pymarc.Record~.

* Features

- Indexing*: Index for both exact and fuzzy matching, useful for looking up identifiers or searching for text
- *Field hooks*: Hook your own Python functions into the indexing of MARC fields. Great for analyzing your collection.
- *Custom charsets*: Optimize memory by adjusting the character set used for indexing.
- *Optimized for MARC binary*: Taking full advantage of the binary MARC format, fastmarc rips through hundreds of thousands of records in the blink of an eye.
- *Compatibility with PyMARC*: Returns ~pymarc.Record~ objects for familiar record access

* Quick Start

#+begin_src python
from fastmarc import MARCReader

with open("records.mrc", "rb") as f:
    reader = (MARCReader(f)
        .add_index("245$a")      # Register field
        .hook("650$a", counter)  # Optional: aggregate
        .index())                # Build index

    results = reader.search("245$a", "music")
    for idx in results[:5]:
        print(reader.get_record(idx)['245']['a'])
#+end_src

* Core Workflow

1. *Open* file with ~MARCReader(fp)~
2. *Register* indexes with ~.add_index()~ and/or hooks with ~.hook()~
3. *Build* with ~.index(charset=...)~
4. *Search* with ~.search()~ or iterate with ~for record in reader~

** Subject Analysis with Hooks

#+begin_src python
from collections import Counter

class SubjectCounter:
    def __init__(self):
        self.counts = Counter()

    def __call__(self, values):
        for v in values:
            self.counts[v.strip()] += 1

subjects = SubjectCounter()

with open("records.mrc", "rb") as f:
    reader = MARCReader(f).hook("650$a", subjects).index()

    print(f"Analyzed {len(reader):,} records")
    print(f"Unique subjects: {len(subjects.counts)}\n")

    for subject, count in subjects.counts.most_common(10):
        print(f"{count:6d}  {subject}")
#+end_src

** Duplicate Detection

#+begin_src python
with open("records.mrc", "rb") as f:
    reader = MARCReader(f).add_index("001", fuzzy=False).index()

    id_index = reader.get_exact_index("001")
    duplicates = {id_: idxs for id_, idxs in id_index.items() if len(idxs) > 1}

    print(f"Found {len(duplicates)} duplicate IDs\n")
    for id_, indices in list(duplicates.items())[:5]:
        print(f"ID {id_}: {len(indices)} records at {indices}")

** Multi-Field Year Extraction

#+begin_src python
from collections import Counter

class YearExtractor:
    def __init__(self):
        self.years = Counter()
        self.source = Counter()

    def __call__(self, fields):
        year = None
        if "008" in fields and fields["008"]:
            year = fields["008"][0][7:11]
            if year.isdigit():
                self.source['008'] += 1
        elif "264$c" in fields:
            import re
            match = re.search(r'(19|20)\d{2}', fields["264$c"][0])
            if match:
                year = match.group(0)
                self.source['264$c'] += 1

        if year and year.isdigit():
            self.years[year] += 1

years = YearExtractor()

with open("records.mrc", "rb") as f:
    reader = MARCReader(f).hook(["008", "264$c"], years).index()

    print(f"Total years: {sum(years.years.values()):,}")
    print(f"Sources: {years.source}")
    print("\nTop 10 years:")
    for year, count in years.years.most_common(10):
        print(f"{year}: {count:6d}")
#+end_src

** Memory-Optimized ISBN Search

#+begin_src python
with open("records.mrc", "rb") as f:
    # Custom charset: only index digits and ISBN chars to optimize memory usage
    reader = (MARCReader(f)
        .add_index("020$a")
        .index(charset="0123456789X-"))

    results = reader.search("020$a", "978-0-")
    print(f"Found {len(results)} records with ISBN prefix '978-0-'")
#+end_src
